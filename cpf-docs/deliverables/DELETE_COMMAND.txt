# Overlay 적용 후 프로젝트 Root PowerShell에서 실행. 사용자 승인된 Root-relative 파일만 제거한다.
$root=(Get-Location).Path; Import-Csv .\cpf-docs\deliverables\DELETE_MANIFEST.csv | Where-Object { $_.approved -eq 'true' -and $_.user_approved -eq 'true' } | ForEach-Object { $rel=$_.path.Replace('/','\'); if($rel -eq 'cpf-docs' -or $rel -like 'cpf-docs\deliverables*' -or $rel -like 'cpf-docs\guides*' -or $rel -like 'cpf-docs\environment\docker*' -or $rel -like 'cpf-tools\environment\docker-development-test*'){ throw "Protected delete path: $rel" }; $p=Join-Path $root $rel; if(Test-Path -LiteralPath $p){ if((Get-Item -LiteralPath $p).PSIsContainer){ throw "Directory delete is not allowed by this manifest command: $rel" }; Remove-Item -LiteralPath $p -Force } }; Write-Host '[CPF][DELETE] 승인된 파일 삭제 완료.'
