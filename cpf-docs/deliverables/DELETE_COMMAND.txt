# Overlay 적용 후 프로젝트 Root PowerShell에서 실행. DELETE_MANIFEST 승인 경로만 제거한다.
$root=(Get-Location).Path; Import-Csv .\cpf-docs\deliverables\DELETE_MANIFEST.csv | Where-Object { $_.approved -eq 'true' } | ForEach-Object { $rel=$_.path.Replace('/','\'); if($rel -eq 'cpf-docs' -or $rel -like 'cpf-docs\deliverables*' -or $rel -like 'cpf-docs\guides*' -or $rel -like 'cpf-docs\environment\docker*'){ throw "Protected delete path: $rel" }; $p=Join-Path $root $rel; if(Test-Path -LiteralPath $p){ Remove-Item -LiteralPath $p -Recurse -Force } }; Write-Host '[CPF][DELETE] approved manifest cleanup completed.'
